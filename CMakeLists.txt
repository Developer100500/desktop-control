cmake_minimum_required(VERSION 3.28)
project(desktop_control VERSION 0.1.3 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(desktop-control src/main.cpp src/common.cpp src/indicator.cpp src/capture.cpp src/uia.cpp src/desktop.cpp src/server.cpp src/foreground.cpp)
add_executable(desktop-control-foreground src/foreground_helper.cpp)
target_include_directories(desktop-control-foreground PRIVATE include third_party)
target_compile_definitions(desktop-control-foreground PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(desktop-control-foreground PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(desktop-control-foreground PRIVATE user32 gdi32)
target_link_options(desktop-control-foreground PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:wmainCRTStartup "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/app.manifest")
target_include_directories(desktop-control PRIVATE include third_party)
target_compile_definitions(desktop-control PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(desktop-control PRIVATE /W4 /EHsc /utf-8 /permissive-)
target_link_libraries(desktop-control PRIVATE windowsapp d3d11 dxgi user32 gdi32 dwmapi ole32 oleaut32 shell32 propsys shlwapi windowscodecs uiautomationcore d2d1 dwrite advapi32)
target_link_options(desktop-control PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/app.manifest")
add_executable(desktop-control-fixture WIN32 tests/fixture.cpp)
add_executable(desktop-control-integration tests/integration.cpp)
target_include_directories(desktop-control-integration PRIVATE third_party include)
target_compile_definitions(desktop-control-integration PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(desktop-control-integration PRIVATE /EHsc /utf-8)
target_link_libraries(desktop-control-integration PRIVATE user32 ole32)
target_link_options(desktop-control-integration PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:wmainCRTStartup "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/app.manifest")
target_compile_definitions(desktop-control-fixture PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(desktop-control-fixture PRIVATE /utf-8)
target_link_libraries(desktop-control-fixture PRIVATE user32 gdi32)
target_link_options(desktop-control-fixture PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/app.manifest")
enable_testing()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
 add_test(NAME mcp_protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_protocol.py $<TARGET_FILE:desktop-control>)
endif()
install(TARGETS desktop-control desktop-control-foreground RUNTIME DESTINATION .)
install(FILES README.md mcp-config.example.json diagnose.ps1 DESTINATION .)
install(FILES third_party/LICENSE.nlohmann-json DESTINATION licenses)

add_executable(desktop-control-timeout-tests tests/test_timeout.cpp src/common.cpp src/desktop.cpp src/foreground.cpp src/uia.cpp src/capture.cpp src/indicator.cpp src/server.cpp)
foreach(property INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS LINK_LIBRARIES LINK_OPTIONS)
  get_target_property(value desktop-control ${property})
  set_property(TARGET desktop-control-timeout-tests PROPERTY ${property} ${value})
endforeach()
add_test(NAME timeout_behavior COMMAND desktop-control-timeout-tests)
